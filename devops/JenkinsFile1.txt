    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building project...'
            }
        }

        stage('Test') {
            steps {
                echo 'Testing project...'
            }
        }

        stage('Generate Artifact') {
            steps {
                echo 'Generating artifact...'
            }
        }
    }

    post {

        }

        failure {

        }
    }
